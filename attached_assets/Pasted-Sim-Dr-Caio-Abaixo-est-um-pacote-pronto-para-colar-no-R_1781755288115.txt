Criar uma tela premium, médica, limpa e TDAH/TOC-friendly para montagem de orçamento. O layout deve usar blocos brancos 3D sobressalentes, sombras suaves e profundas, bordas arredondadas e hierarquia visual clara. O fundo geral deve ser claro.   label: string;
  value: string;
  tone?: "default" | "danger" | "success";
  icon?: string;
};

function SummaryCard({ label, value, tone = "default", icon }: SummaryCardProps) {
  return (
    <div className={`summary-card summary-card--${tone}`}>
      <span className="summary-card__label">{label}</span>
      <strong className="summary-card__value">
        {icon && <span className="summary-card__icon">{icon}</span>}
        {value}
      </strong>
    </div>
  );
}

function SessionMeter() {
  const bars = [
    "high",
    "high",
    "high",
    "medium",
    "medium",
    "warning",
    "warning",
    "danger",
    "empty",
    "empty",
    "empty",
    "empty",
  ];

  return (
    <div className="session-meter" aria-label="Indicador visual de sessões">
      {bars.map((status, index) => (
        <span key={index} className={`session-meter__bar session-meter__bar--${status}`} />
      ))}
    </div>
  );
}

function ProgressSegments() {
  return (
    <div className="volume-progress" aria-label="Progresso para próxima faixa de desconto">
      {Array.from({ length: 10 }).map((_, index) => (
        <span
          key={index}
          className={
            index < 5
              ? "volume-progress__segment volume-progress__segment--active"
              : "volume-progress__segment"
          }
        />
      ))}
    </div>
  );
}
